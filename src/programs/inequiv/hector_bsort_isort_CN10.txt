def hector_bsort_isort_CN10(args):
    compare = args[0]
    read_x = args[1][0]
    write_x = args[1][1]

    length = 10
    a = [0] * 10

    def a_get(n):
        if 0 <= n < 10:
            return a[n]
        return -1

    def a_set(n, m):
        if 0 <= n < 10:
            a[n] = m

    i = 0
    while i < length:
        a_set(i, read_x())
        i += 1

    i = 1
    while i < length:
        val = a_get(i)
        j = i
        while j > 0 and compare(a_get(j - 1), val):
            a_set(j, a_get(j - 1))
            j -= 1
        a_set(j, val)
        i += 1

    i = 0
    while i < length:
        write_x(a_get(i))
        i += 1
