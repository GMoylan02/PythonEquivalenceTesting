x = 0

def call_even():
    global x
    # parity test: x mod 2 == 0
    if (x - ((x // 2) * 2)) == 0:
        x = x + 1
    return x < 5

def call_odd():
    global x
    if (x - ((x // 2) * 2)) != 0:
        x = x + 1
    return x < 5

def program(f):
    # f : (unit -> bool) -> (unit -> bool) -> unit
    return f(call_even, call_odd)
