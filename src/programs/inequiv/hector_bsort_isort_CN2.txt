    compare = args[0]
    read_x = args[1][0]
    write_x = args[1][1]

    a = [0, 0]

    def a_get(n):
        if n == 0:
            return a[0]
        elif n == 1:
            return a[1]
        return -1

    def a_set(n, m):
        if n == 0:
            a[0] = m
        elif n == 1:
            a[1] = m

    i = 0
    while i < 2:
        val = read_x()
        a_set(i, val)
        i += 1

    i = 1
    while i < 2:
        val = a_get(i)
        j = i

        while j > 0 and compare(a_get(j - 1), val):
            a_set(j, a_get(j - 1))
            j -= 1

        a_set(j, val)
        i += 1

    i = 0
    while i < 2:
        write_x(a_get(i))
        i += 1
