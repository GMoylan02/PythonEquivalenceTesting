    if n == 0: return a0

    elif n == 7: return a7
    elif n == 8: return a8
    elif n == 9: return a9
    else: return -1


def bsearch(key):
    global lo, hi, found

    if lo + 1 < hi:
        mid = (lo + hi) // 2

        if get(mid) <= key:
            lo = mid + 1
        else:
            hi = mid

        return bsearch(key)
    else:
        found = (get(lo) == key)
        return found
